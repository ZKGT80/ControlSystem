#include <Wire.h>
#include <LiquidCrystal_I2C.h>

#define TRIG_PIN 23
#define ECHO_PIN 19
#define BUZZER_PIN 13

// LEDs
#define GREEN_LED 32
#define YELLOW_LED 26
#define RED_LED 12

LiquidCrystal_I2C lcd(0x27, 16, 2);

#define BUZZER_FREQ 1000
#define BUZZER_RESOLUTION 8

void setup() {

  Serial.begin(115200);

  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);

  // LED pins
  pinMode(GREEN_LED, OUTPUT);
  pinMode(YELLOW_LED, OUTPUT);
  pinMode(RED_LED, OUTPUT);

  lcd.init();
  lcd.backlight();

  lcd.setCursor(0,0);
  lcd.print("SmartDesk AI");

  lcd.setCursor(0,1);
  lcd.print("Starting...");

  delay(2000);
  lcd.clear();

  // Buzzer setup (ESP32 v3.x)
  ledcAttach(BUZZER_PIN, BUZZER_FREQ, BUZZER_RESOLUTION);
}

void loop() {

  // Trigger ultrasonic
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  long duration = pulseIn(ECHO_PIN, HIGH, 30000);
  float distance = duration * 0.0343 / 2;

  Serial.println(distance);

  lcd.clear();

  // -------------------------
  // RESET ALL OUTPUTS FIRST
  // -------------------------
  digitalWrite(GREEN_LED, LOW);
  digitalWrite(YELLOW_LED, LOW);
  digitalWrite(RED_LED, LOW);
  ledcWriteTone(BUZZER_PIN, 0);

  // -------------------------
  // CONDITIONS
  // -------------------------

  // No sensor
  if (duration == 0) {

    lcd.setCursor(0,0);
    lcd.print("No Sensor");
  }

  // DANGER ZONE
  else if (distance < 15) {

    digitalWrite(RED_LED, HIGH);
    ledcWriteTone(BUZZER_PIN, BUZZER_FREQ);

    lcd.setCursor(0,0);
    lcd.print("MOVE BACK");
    lcd.setCursor(0,1);
    lcd.print(distance);
    lcd.print(" cm");
  }

  // Close Distance
  else if (distance >= 15 && distance <= 25) {

    digitalWrite(YELLOW_LED, HIGH);

    lcd.setCursor(0,0);
    lcd.print("Close Distance");
    lcd.setCursor(0,1);
    lcd.print(distance);
    lcd.print(" cm");
  }

  // Good Distance
  else if (distance > 25 && distance <= 40) {

    digitalWrite(GREEN_LED, HIGH);

    lcd.setCursor(0,0);
    lcd.print("Good Distance");
    lcd.setCursor(0,1);
    lcd.print(distance);
    lcd.print(" cm");
  }

  // NO USER
  else {

    lcd.setCursor(0,0);
    lcd.print("NO USER");
  }

  delay(800);
}
